-- 3. CREATE TABLE: gs_about_content
create table if not exists public.gs_about_content (
  id uuid primary key default gen_random_uuid(),
  section_key text not null unique, -- 'description', 'background', 'address', 'team_intro'
  content_text text not null,
  updated_at timestamptz default now(),
  updated_by uuid references public.gs_admins(id)
);

-- 4. CREATE TABLE: gs_team_members
create table if not exists public.gs_team_members (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  role text not null, -- e.g. 'Creative Director', 'Program Director'
  bio text,
  photo_url text,
  is_partner boolean default false, -- true = Partner org (ESGMC, Smart ICT), false = core team
  display_order int default 0,
  updated_at timestamptz default now()
);

-- 5. CREATE TABLE: gs_how_it_works
create table if not exists public.gs_how_it_works (
  id uuid primary key default gen_random_uuid(),
  section_key text not null unique,
  -- 'intro', 'skills', 'institute_programs', 'progression', 'marketplace',
  -- 'shadow_framework', 'bridge', 'trainer_track', 'coming_soon'
  content_text text not null,
  updated_at timestamptz default now(),
  updated_by uuid references public.gs_admins(id)
);

-- 6. ENABLE ROW LEVEL SECURITY (RLS) & POLICIES
alter table public.gs_about_content enable row level security;
alter table public.gs_team_members enable row level security;
alter table public.gs_how_it_works enable row level security;

-- Public read access
drop policy if exists "Allow public read on gs_about_content" on public.gs_about_content;
create policy "Allow public read on gs_about_content" on public.gs_about_content for select using (true);

drop policy if exists "Allow public read on gs_team_members" on public.gs_team_members;
create policy "Allow public read on gs_team_members" on public.gs_team_members for select using (true);

drop policy if exists "Allow public read on gs_how_it_works" on public.gs_how_it_works;
create policy "Allow public read on gs_how_it_works" on public.gs_how_it_works for select using (true);

-- Allow anon/authenticated upsert & update
drop policy if exists "Allow anon insert/update on gs_about_content" on public.gs_about_content;
create policy "Allow anon insert/update on gs_about_content" on public.gs_about_content for all using (true) with check (true);

drop policy if exists "Allow anon insert/update on gs_team_members" on public.gs_team_members;
create policy "Allow anon insert/update on gs_team_members" on public.gs_team_members for all using (true) with check (true);

drop policy if exists "Allow anon insert/update on gs_how_it_works" on public.gs_how_it_works;
create policy "Allow anon insert/update on gs_how_it_works" on public.gs_how_it_works for all using (true) with check (true);
